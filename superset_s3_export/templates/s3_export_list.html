(function() { let datasets = []; let currentDataset = null; async function loadDatasets() { try { const response = await fetch("/api/v1/s3-export/datasets"); if (!response.ok) throw new Error(`HTTP ${response.status}`); datasets.forEach((dataset) => { const option = document.createElement("option"); option.value = dataset.id; option.textContent = dataset.name; select.appendChild(option); }); } catch (error) { console.error("Failed to load datasets:", error); alert("Failed to load: " + error.message); } } function updateFilters() { const datasetId = document.getElementById("datasetSelect").value; const filtersContainer = document.getElementById("filtersContainer"); if (!currentDataset) return; descDiv.textContent = currentDataset.description || ""; const regionSelect = document.getElementById("regionSelect"); regionSelect.innerHTML = ''; currentDataset.filters.regions.forEach((region) => { const option = document.createElement("option"); option.value = region; option.textContent = region; regionSelect.appendChild(option); }); const today = new Date().toISOString().split("T")[0]; const lastMonth = new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString().split("T")[0]; document.getElementById("endDate").value = today; document.getElementById("startDate").value = lastMonth; filtersContainer.style.display = "block"; } async function submitExport() { const email = document.getElementById("emailInput").value; const startDate = document.getElementById("startDate").value; const endDate = document.getElementById("endDate").value; const region = document.getElementById("regionSelect").value; if (!email || !startDate || !endDate || !region) { alert("Please fill in all fields"); return; } try { const response = await fetch("/api/v1/s3-export/create", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ dataset_id: currentDataset.id, filters: { start_date: startDate, end_date: endDate, region: region }, email: email, }), }); if (response.ok) { alert("✓ Export started! Check your email shortly."); document.getElementById("emailInput").value = ""; } else { const error = await response.json(); alert("Failed: " + (error.message || "Unknown error")); } } catch (error) { alert("Failed: " + error.message); } } document.addEventListener('DOMContentLoaded', function() { loadDatasets(); document.getElementById('datasetSelect').addEventListener('change', updateFilters); document.getElementById('exportBtn').addEventListener('click', submitExport); }); })();
