Export Large Datasets to S3

Select a dataset, apply filters, and receive a download link via email.

let datasets = []; let currentDataset = null; async function loadDatasets() { try { const response = await fetch("/api/v1/s3-export/datasets"); if (!response.ok) { throw new Error(`HTTP ${response.status}`); } const data = await response.json(); datasets = data.datasets || []; if (datasets.length === 0) { alert( "No export datasets configured. Please contact your administrator.", ); return; } const select = document.getElementById("datasetSelect"); datasets.forEach((dataset) => { const option = document.createElement("option"); option.value = dataset.id; option.textContent = dataset.name; select.appendChild(option); }); } catch (error) { console.error("Failed to load datasets:", error); alert( "Failed to load export configurations: " + error.message, ); } } function updateFilters() { const datasetId = document.getElementById("datasetSelect").value; const filtersContainer = document.getElementById("filtersContainer"); const descDiv = document.getElementById("datasetDesc"); if (!datasetId) { filtersContainer.style.display = "none"; descDiv.textContent = ""; currentDataset = null; return; } currentDataset = datasets.find((d) => d.id === datasetId); if (!currentDataset) return; descDiv.textContent = currentDataset.description || ""; const regionSelect = document.getElementById("regionSelect"); regionSelect.innerHTML = ''; currentDataset.filters.regions.forEach((region) => { const option = document.createElement("option"); option.value = region; option.textContent = region; regionSelect.appendChild(option); }); const today = new Date().toISOString().split("T")[0]; const lastMonth = new Date( Date.now() - 30 * 24 * 60 * 60 * 1000, ) .toISOString() .split("T")[0]; document.getElementById("endDate").value = today; document.getElementById("startDate").value = lastMonth; filtersContainer.style.display = "block"; } async function submitExport() { const email = document.getElementById("emailInput").value; const startDate = document.getElementById("startDate").value; const endDate = document.getElementById("endDate").value; const region = document.getElementById("regionSelect").value; if (!email) { alert("Please enter your email"); return; } if (!startDate || !endDate) { alert("Please select date range"); return; } if (!region) { alert("Please select a region"); return; } try { const response = await fetch("/api/v1/s3-export/create", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ dataset_id: currentDataset.id, filters: { start_date: startDate, end_date: endDate, region: region, }, email: email, }), }); if (response.ok) { alert( "✓ Export started! You will receive an email with the download link shortly.", ); document.getElementById("emailInput").value = ""; } else { const error = await response.json(); alert( "Export failed: " + (error.message || "Unknown error"), ); } } catch (error) { console.error("Export error:", error); alert("Export failed: " + error.message); } } loadDatasets();
